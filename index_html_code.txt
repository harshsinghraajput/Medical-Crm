  <script>
    // =========================================================================
    // AI-First CRM - Frontend JavaScript
    // =========================================================================
    // This is the main frontend code that handles:
    // - User interaction with the chat interface
    // - Sending messages to the backend API
    // - Auto-filling the form based on AI responses
    // - Quick edit functionality (click buttons or type numbers 1-9)
    // - Auto-save after 1 minute of inactivity
    // - Success popup animation
    //
    // HOW IT WORKS:
    // 1. User types message in chat input
    // 2. Frontend sends to /api/chat endpoint
    // 3. Backend returns response + form_data
    // 4. Frontend displays response and auto-fills form
    // 5. Quick edit bar appears for easy field editing
    //
    // QUICK EDIT FLOW:
    // - Type "1" → placeholder changes to "Type new value for HCP Name..."
    // - Type the value → sends "change hcp_name to [value]"
    // - Form updates instantly
    // =========================================================================
    (function() {
      // API URL - adapts to current host (works on any port)
      var API = window.location.origin;
      // Session ID - unique for each browser tab
      var SID = "session-" + Date.now();
      // Loading state - prevents multiple simultaneous requests
      var loading = false;
      // Current field being edited - null when not editing
      var editingField = null;

      // =========================================================================
      // FIELD MAPPING: Maps backend field names to HTML element IDs
      // Used to auto-fill the form when backend returns form_data
      // =========================================================================
      var FIELDS = {
        hcp_name: "f_name", specialty: "f_specialty", hospital: "f_hospital",
        interaction_type: "f_type", date: "f_date", sentiment: "f_sentiment",
        topics_discussed: "f_topics", materials_shared: "f_materials",
        follow_up_actions: "f_followup", notes: "f_notes"
      };

      var FIELD_NAMES = {
        "1": "hcp_name", "2": "specialty", "3": "hospital", "4": "interaction_type",
        "5": "date", "6": "sentiment", "7": "topics_discussed", "8": "materials_shared",
        "9": "follow_up_actions"
      };

      var FIELD_LABELS = {
        hcp_name: "HCP Name", specialty: "Specialty", hospital: "Hospital",
        interaction_type: "Interaction Type", date: "Date", sentiment: "Sentiment",
        topics_discussed: "Topics", materials_shared: "Materials",
        follow_up_actions: "Follow-up"
      };

      // Idle timer for auto-save after 1 minute
      var idleTimer = null;
      function startIdleTimer() {
        stopIdleTimer();
        idleTimer = setTimeout(function() {
          showSuccess();
        }, 60000); // 60 seconds = 1 minute
      }
      function stopIdleTimer() {
        if (idleTimer) { clearTimeout(idleTimer); idleTimer = null; }
      }

      function $(id) { return document.getElementById(id); }

      function setForm(data) {
        if (!data) return;
        var keys = Object.keys(FIELDS);
        for (var i = 0; i < keys.length; i++) {
          var k = keys[i];
          if (data[k] !== undefined && data[k] !== null) {
            var el = $(FIELDS[k]);
            if (el) el.value = data[k];
          }
        }
      }

      function clearForm() {
        var ids = Object.values(FIELDS);
        for (var i = 0; i < ids.length; i++) { var el = $(ids[i]); if (el) el.value = ""; }
        $("formStatus").className = "form-msg info";
        $("formStatus").textContent = "Form cleared";
        $("submitBtn").style.display = "none";
        editingField = null;
      }

      function addMsg(role, content, toolCalls, extra) {
        var box = $("chatMessages");
        var w = box.querySelector(".welcome");
        if (w) w.remove();
        var div = document.createElement("div");
        div.className = "msg " + (role === "user" ? "user" : "bot");
        var html = '<div class="bubble">' + escHtml(content) + '</div>';
        if (toolCalls && toolCalls.length > 0) html += '<div class="tool-tag">Tools: ' + toolCalls.join(", ") + '</div>';
        if (extra) html += extra;
        html += '<div class="meta">' + (role === "user" ? "You" : "AI Assistant") + '</div>';
        div.innerHTML = html;
        box.appendChild(div);
        box.scrollTop = box.scrollHeight;
        return div;
      }

      function showTyping() {
        var box = $("chatMessages");
        var div = document.createElement("div");
        div.className = "msg bot"; div.id = "typingMsg";
        div.innerHTML = '<div class="bubble"><div class="typing"><div class="dot"></div><div class="dot"></div><div class="dot"></div></div></div>';
        box.appendChild(div); box.scrollTop = box.scrollHeight;
      }
      function hideTyping() { var t = $("typingMsg"); if (t) t.remove(); }
      function escHtml(s) { return s.replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/\n/g,"<br>"); }

      function showSuccess() { $("successPopup").classList.add("active"); }

      // =====================================================================
      // EDIT FLOW: Click field buttons to edit directly
      // =====================================================================
      function askWhatToChange() {
        editingField = null;
        var extra = '<div class="choice-row">' +
          '<button class="choice-btn yes" data-field="hcp_name">1. Name</button>' +
          '<button class="choice-btn yes" data-field="specialty">2. Specialty</button>' +
          '<button class="choice-btn yes" data-field="hospital">3. Hospital</button>' +
          '<button class="choice-btn yes" data-field="interaction_type">4. Type</button>' +
          '<button class="choice-btn yes" data-field="sentiment">5. Sentiment</button>' +
          '<button class="choice-btn yes" data-field="topics_discussed">6. Topics</button>' +
          '<button class="choice-btn yes" data-field="materials_shared">7. Materials</button>' +
          '<button class="choice-btn yes" data-field="follow_up_actions">8. Follow-up</button>' +
          '<button class="choice-btn no" data-field="submit">Submit</button></div>';
        var div = addMsg("assistant", "Click a field to edit:", null, extra);
        bindFieldBtns(div);
      }

      function bindFieldBtns(container) {
        var btns = container.querySelectorAll(".choice-btn");
        for (var i = 0; i < btns.length; i++) {
          btns[i].addEventListener("click", function() {
            stopIdleTimer();
            var field = this.getAttribute("data-field");
            if (field === "submit") { showSuccess(); return; }
            editingField = field;
            $("chatInput").placeholder = "Type new value for " + (FIELD_LABELS[field] || field) + "...";
            $("chatInput").focus();
          });
        }
      }

      // Quick edit bar buttons
      function bindQuickEdit() {
        var btns = document.querySelectorAll("[data-qe]");
        for (var i = 0; i < btns.length; i++) {
          btns[i].addEventListener("click", function() {
            stopIdleTimer();
            var field = this.getAttribute("data-qe");
            if (field === "submit") { showSuccess(); return; }
            var fieldKey = FIELD_NAMES[field] || field;
            editingField = fieldKey;
            $("chatInput").placeholder = "Type new value for " + (FIELD_LABELS[fieldKey] || fieldKey) + "...";
            $("chatInput").focus();
          });
        }
      }
      bindQuickEdit();

      // Show quick edit after logging
      function showQuickEdit() { $("quickEdit").style.display = "block"; }
      function hideQuickEdit() { $("quickEdit").style.display = "none"; }

      // =====================================================================
      // SEND MESSAGE: Handle user input
      // =====================================================================
      // This is the main function that handles all user input.
      //
      // FLOW:
      // 1. User types message and presses Enter or clicks Send
      // 2. Check if it's a number (1-9) → quick edit mode
      // 3. Check if we're editing a field → send "change X to Y"
      // 4. Otherwise → send as normal message to backend
      // 5. Backend responds with form_data → auto-fill form
      //
      // QUICK EDIT:
      // - User types "6" → editingField = "sentiment"
      // - Placeholder changes to "Type new value for Sentiment..."
      // - User types "Positive" → sends "change sentiment to Positive"
      // - Form updates instantly
      // =====================================================================
      function sendToBackend(msg, cb) {
        showTyping();
        var xhr = new XMLHttpRequest();
        xhr.open("POST", API + "/api/chat", true);
        xhr.setRequestHeader("Content-Type", "application/json");
        xhr.timeout = 30000;
        xhr.onload = function() { hideTyping(); try { cb(JSON.parse(xhr.responseText)); } catch(e) { cb(null); } };
        xhr.onerror = function() { hideTyping(); addMsg("assistant", "Cannot connect."); cb(null); };
        xhr.ontimeout = function() { hideTyping(); addMsg("assistant", "Timed out."); cb(null); };
        xhr.send(JSON.stringify({ message: msg, session_id: SID }));
      }

      function sendMsg() {
        var input = $("chatInput");
        var msg = input.value.trim();
        if (!msg || loading) return;
        loading = true; $("sendBtn").disabled = true;
        addMsg("user", msg); input.value = "";

        // =================================================================
        // QUICK EDIT: Handle number input (1-9)
        // When user types a number, switch to edit mode for that field
        // =================================================================
        if (!editingField && /^[1-9]$/.test(msg)) {
          var fieldKey = FIELD_NAMES[msg];
          if (fieldKey) {
            editingField = fieldKey;
            loading = false; $("sendBtn").disabled = false;
            $("chatInput").value = "";
            $("chatInput").placeholder = "Type new value for " + (FIELD_LABELS[fieldKey] || fieldKey) + "...";
            $("chatInput").focus();
            return;
          }
        }

        // =================================================================
        // EDIT MODE: If we're editing a field, send as edit command
        // Converts user input to "change [field] to [value]" format
        // =================================================================
        if (editingField) {
          var editMsg = "change " + editingField + " to " + msg;
          editingField = null;
          $("chatInput").placeholder = "Describe your HCP interaction...";
          sendToBackend(editMsg, function(data) {
            loading = false; $("sendBtn").disabled = false;
            if (!data) return;
            addMsg("assistant", data.response, data.tool_calls);
            if (data.form_data) setForm(data.form_data);
            $("formStatus").className = "form-msg ok";
            $("formStatus").textContent = "Form updated by AI";
            $("submitBtn").style.display = "inline-block";
            startIdleTimer();
          });
          return;
        }

        sendToBackend(msg, function(data) {
          loading = false; $("sendBtn").disabled = false;
          if (!data) return;

          var didLog = false;
          var didEdit = false;
          if (data.tool_calls) {
            for (var i = 0; i < data.tool_calls.length; i++) {
              if (data.tool_calls[i].indexOf("log_interaction") !== -1) didLog = true;
              if (data.tool_calls[i].indexOf("edit_interaction") !== -1) didEdit = true;
            }
          }

          if (didLog) {
            addMsg("assistant", data.response, data.tool_calls);
            if (data.form_data) {
              setForm(data.form_data);
              $("formStatus").className = "form-msg ok";
              $("formStatus").textContent = "Form updated by AI";
              $("submitBtn").style.display = "inline-block";
            }
            showQuickEdit();
            startIdleTimer(); // Start 1-minute auto-save timer
          } else if (didEdit) {
            addMsg("assistant", data.response, data.tool_calls);
            if (data.form_data) setForm(data.form_data);
            startIdleTimer(); // Reset timer on edit
          } else {
            addMsg("assistant", data.response, data.tool_calls);
            if (data.form_data) setForm(data.form_data);
          }
        });
      }

      $("sendBtn").addEventListener("click", sendMsg);
      $("chatInput").addEventListener("keydown", function(e) { if (e.key === "Enter") sendMsg(); });
      $("clearBtn").addEventListener("click", function() { clearForm(); stopIdleTimer(); hideQuickEdit(); });
      $("submitBtn").addEventListener("click", function() { stopIdleTimer(); hideQuickEdit(); showSuccess(); });
      $("okBtn").addEventListener("click", function() {
        stopIdleTimer(); hideQuickEdit();
        $("successPopup").classList.remove("active"); clearForm();
        $("chatMessages").innerHTML = '<div class="welcome"><div class="icon">&#9993;</div><h3>Welcome to AI CRM Assistant</h3><p>Describe your HCP interaction in natural language</p><div style="margin-top:16px; text-align:left; max-width:500px; margin-left:auto; margin-right:auto;"><div style="font-size:0.72rem; color:var(--text-dim); margin-bottom:6px; text-transform:uppercase; letter-spacing:0.5px;">Try saying something like:</div><div style="background:var(--bg-card); border:1px solid var(--border); border-radius:10px; padding:12px 16px; font-size:0.82rem; color:var(--text); line-height:1.7;"><div style="color:var(--accent); font-weight:600; margin-bottom:4px;">Template:</div>Met with <span style="color:var(--green);">Dr. [Name]</span>, <span style="color:var(--green);">[Specialty]</span> at <span style="color:var(--green);">[Hospital]</span>. Discussed <span style="color:var(--green);">[topics]</span>. Sentiment: <span style="color:var(--green);">[Positive/Neutral/Negative]</span>.</div><div style="font-size:0.72rem; color:var(--text-dim); margin-top:8px;">Example: "Met with Dr. Sarah, Cardiology at City Hospital. Discussed new treatment plan. Very positive meeting."</div></div></div>';
        bindSugBtns();
      });

      // Clear form on page load
      clearForm();
      // Reset backend form state
      var xhr = new XMLHttpRequest();
      xhr.open("POST", API + "/api/form/reset", true);
      xhr.setRequestHeader("Content-Type", "application/json");
      xhr.send();

    })();
